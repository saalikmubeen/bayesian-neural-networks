class DNN(nn.Module):
    """Standard two-hidden-layer MLP."""
    def __init__(self, hidden=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Flatten(),
            nn.Linear(784, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden), nn.ReLU(),
            nn.Linear(hidden, 10),
        )

    def forward(self, x):
        return self.net(x)


def train_dnn(model, loader, epochs=20, lr=1e-3):
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    model.train()
    for epoch in range(epochs):
        total = 0
        for X, Y in loader:
            x = X.to(device)
            y = Y.to(device)
            out = model(x)
            optimizer.zero_grad()
            loss = F.cross_entropy(out, y)
            loss.backward()
            optimizer.step()
            total += loss.item()

        avg_loss = total / len(loader)
        if (epoch + 1) % 5 == 0:
            print(f"  [DNN] epoch {epoch+1:3d}  loss={avg_loss:.4f}")


@torch.no_grad()
def eval_dnn(model, loader):
    model.eval()
    predicted_probs, labels = [], []
    for X, Y in loader:
        out = model(X.to(device))
        probs = F.softmax(out, dim=-1)
        predicted_probs.append(probs.cpu())
        labels.append(Y)
    return torch.cat(predicted_probs), torch.cat(labels)

3. Bayesian Neural Network — Bayes by Backprop¶

3.1 Probabilistic Setup¶

Each weight $w$ has a posterior $q(w|\mathcal{D}) = \mathcal{N}(\mu, \sigma^2)$
and a prior $p(w) = \mathcal{N}(0, 1)$.

Then optimize the ELBO:
$$\mathcal{L} = \underbrace{\mathbb{E}_{q}[\log p(\mathcal{D}|w)]}_{\text{likelihood}} - \underbrace{\text{KL}[q(w|\mathcal{D}) \| p(w)]}_{\text{complexity penalty}}$$

ELBO loss = CE loss + β · KL, where β = 1/N scales the KL to match the per-sample likelihood.

3.2 Reparameterization Trick¶

To backpropagate through stochastic weights, I used:
$$w = \mu + \sigma \cdot \epsilon, \quad \epsilon \sim \mathcal{N}(0, I)$$

And parameterized $\sigma = \text{softplus}(\rho) = \log(1 + e^\rho)$ to keep $\sigma > 0$.

3.3 Prior Choice¶

Standard N(0,1) prior

class BayesianLinear(nn.Module):

    def __init__(self, in_features, out_features):
        super().__init__()
        # Weight posterior params
        # Init mu ~ N(0, 0.1), rho = -3 => sigma ≈ 0.048
        self.mu  = nn.Parameter(torch.randn(out_features, in_features) * 0.1)

        # sigma(Standard deviation) cant't be negative, so we use rho as a raw parameter
        # and apply softplus to get sigma
        self.rho = nn.Parameter(torch.full((out_features, in_features), -3.0))

        # Bias posterior params
        self.bias_mu  = nn.Parameter(torch.zeros(out_features))
        self.bias_rho = nn.Parameter(torch.full((out_features,), -3.0))

    @property
    def sigma_w(self):
        return F.softplus(self.rho)

    @property
    def sigma_b(self):
        return F.softplus(self.bias_rho)

    def forward(self, x):
        # Create a matrix of the same shape as self.mu
        # Each value in that matrix is randomly drawn from N(0, 1) (standard normal distribution)
        # Mean = 0, Standard deviation = 1
        eps_w = torch.randn_like(self.mu)
        eps_b = torch.randn_like(self.bias_mu)

        # Reparameterization: w = mu + sigma * eps,  eps ~ N(0,1)
        # Any normal distribution can be created from the standard normal N(0,1):
        # Random number from N(μ, σ²) = μ + σ × (random number from N(0,1))
        w = self.mu + self.sigma_w * eps_w
        b = self.bias_mu + self.sigma_b * eps_b
        return F.linear(x, w, b)

    def kl_divergence(self):
        # Prior: N(0, 1) on all weights and biases
        # KL = log(σ₂/σ₁) + (σ₁² + (μ₁ - μ₂)²) / (2σ₂²) - 1/2
        kl_w = 0.5 * (self.sigma_w**2 + self.mu**2 - 1 - torch.log(self.sigma_w**2 + 1e-8))
        kl_b = 0.5 * (self.sigma_b**2 + self.bias_mu**2 - 1 - torch.log(self.sigma_b**2 + 1e-8))
        return kl_w.sum() + kl_b.sum()


class BayesianNN(nn.Module):
    """Three-layer BNN: 784 -> 128 -> 128 -> 10."""

    def __init__(self, hidden=128):
        super().__init__()
        self.fc1 = BayesianLinear(784, hidden)
        self.fc2 = BayesianLinear(hidden, hidden)
        self.fc3 = BayesianLinear(hidden, 10)

    def forward(self, x):
        x = x.view(-1, 784)
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.fc3(x)   # logits

    def kl_total(self):
        return self.fc1.kl_divergence() + self.fc2.kl_divergence() + self.fc3.kl_divergence()

def train_bnn(model, loader, epochs=30, lr=1e-3):

    opt = torch.optim.Adam(model.parameters(), lr=lr)
    kl_weight = 1.0 / len(loader.dataset)
    model.train()

    history = {"elbo": [], "ce": [], "kl": []}

    for epoch in range(epochs):
        total_elbo = 0
        total_ce = 0
        total_kl = 0
        for X, Y in loader:
            x, y = X.to(device), Y.to(device)
            opt.zero_grad()

            logits  = model(x)
            ce_loss = F.cross_entropy(logits, y)
            kl      = model.kl_total()
            loss    = ce_loss + kl_weight * kl

            loss.backward()
            opt.step()

            total_elbo += loss.item()
            total_ce   += ce_loss.item()
            total_kl   += kl.item()

        n = len(loader)
        history["elbo"].append(total_elbo / n)
        history["ce"].append(total_ce / n)
        history["kl"].append(total_kl / n)

        if (epoch + 1) % 5 == 0:
            print(f"  [BNN] epoch {epoch+1:3d}  "
                  f"ELBO={total_elbo/n:.4f}  "
                  f"CE={total_ce/n:.4f}  "
                  f"KL={total_kl/n:.1f}")

    return history



# Run the network many times with different random weights, average the results.
def predict_bnn(model, loader, n_samples=50):

    model.eval()
    all_predictions = []
    labels = []

    for X, Y in loader:
        x = X.to(device)
        y = Y.to(device)

        # List of n_samples tensors, each of shape (batch_size, 10)
        batch_predictions = []  # stores predictions for THIS batch across all MC samples
        with torch.no_grad():
            for _ in range(n_samples):
                # Each forward pass draws fresh weight samples → different prediction
                logits = model(x)  # Each call generates NEW random weights
                out = F.softmax(logits, dim=-1)
                batch_predictions.append(out.cpu())
        batch_tensor = torch.stack(batch_predictions, dim=0) # (n_samples, batch, 10)
        all_predictions.append(batch_tensor)
        labels.append(y.cpu())

    all_predictions = torch.cat(all_predictions, dim=1)  # (n_samples, N, 10)
    labels = torch.cat(labels, dim=0)  # (N,)

    mean_probs = all_predictions.mean(0)  # (N, 10) # Average probability across MC samples for each class
    std_probs = all_predictions.std(0)  # (N, 10)
    return mean_probs, std_probs, labels

5. Uncertainty Metrics¶

  • Max confidence: $\max_k p(y=k|x)$ | overconfidence indicator
  • Predictive entropy: $H[y|x] = -\sum_k \bar{p}_k \log \bar{p}_k$ | total uncertainty
  • Epistemic uncertainty: mean std across MC samples | model uncertainty
In [ ]:
def predictive_entropy(mean_probs, eps=1e-8):
    # H[y|x] = -sum p*log(p). High => uncertain.
    return -(mean_probs * (mean_probs + eps).log()).sum(-1)

def max_confidence(probs):

    return probs.max(-1).values

ENTROPY_THRESHOLD = 0.5 # If entropy > threshold, reject the prediction as too uncertain.


def predict_bnn_with_rejection(
    model, loader, n_samples=50, threshold=ENTROPY_THRESHOLD
):
    mean_probs, std_probs, labels = predict_bnn(model, loader, n_samples)
    entropy = predictive_entropy(mean_probs)

    accepted_mask = entropy < threshold
    accepted_preds = mean_probs[accepted_mask].argmax(-1)
    accepted_labels = labels[accepted_mask]
    acc_accepted = (
        accuracy_score(accepted_labels, accepted_preds)
        if len(accepted_labels) > 0
        else 0
    )

    return {
        "accept_rate": accepted_mask.float().mean().item(),
        "reject_rate": (1 - accepted_mask.float().mean()).item(),
        "accuracy_on_accepted": acc_accepted,
        "entropy": entropy.numpy(),
        "mask": accepted_mask,
    }


def predict_dnn_with_rejection(model, loader, threshold=ENTROPY_THRESHOLD):
    probs, labels = eval_dnn(model, loader)
    confidence = max_confidence(probs)  # (N,)
    uncertainty = 1 - confidence  # lower confidence = more uncertain

    accepted_mask = uncertainty < threshold  # reject if uncertainty > threshold
    accepted_preds = probs[accepted_mask].argmax(-1)
    accepted_labels = labels[accepted_mask]
    acc_accepted = (
        accuracy_score(accepted_labels, accepted_preds)
        if len(accepted_labels) > 0
        else 0
    )

    return {
        "accept_rate": accepted_mask.float().mean().item(),
        "reject_rate": (1 - accepted_mask.float().mean()).item(),
        "accuracy_on_accepted": acc_accepted,
        "uncertainty": uncertainty.numpy(),
        "mask": accepted_mask,
    }
